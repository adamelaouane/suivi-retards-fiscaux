## 5. RÈGLES MÉTIER (cahier des charges)
- Montant restant = montant_exigible − total des paiements VALIDÉS ; jamais négatif.
- Statut : PAYE (restant = 0) ; PARTIELLEMENT_PAYE (payé > 0 et restant > 0) ; IMPAYE (payé = 0 et restant > 0).
- EN RETARD est un indicateur SÉPARÉ du statut : date_limite < aujourd'hui ET restant > 0.
  Échéance aujourd'hui = pas en retard. jours_retard = (aujourd'hui − date_limite).days.
  Un dossier sans restant n'est jamais impayé ni en retard.
- Paiement : montant > 0, dossier existant, total payé ≤ montant dû, référence unique si fournie,
  enregistrement transactionnel (tout ou rien). Pas de suppression définitive d'un paiement validé :
  annulation/correction tracée.
- Unicité : identifiant_fiscal, numero_dossier, TypeTaxe.code, référence de paiement.
- Un type de taxe utilisé n'est jamais supprimé : on le désactive (actif = False).
- Seuils de retard (significatif / critique), lignes par page, etc. : configurables via le modèle Parametre
  (pas de valeurs en dur dispersées).
- Import : validation avant intégration, doublons détectés, lignes acceptées/rejetées distinguées,
  jamais d'import silencieux de données invalides.
- Journal d'audit : action, date/heure, objet concerné, description, utilisateur (quand l'auth existera) ;
  protégé contre la modification ; jamais de secrets dedans.
- Exports : PDF (consultation/impression) et Excel (analyse) ; indiquer filtres utilisés, date de génération
  et le caractère démonstratif des données.

import os

from flask import Flask

from app.extensions import db, migrate


def create_app(config=None):
    app = Flask(__name__, instance_relative_config=True)

    app.config["SECRET_KEY"] = os.getenv(
        "SECRET_KEY",
        "dev-key-change-me"
    )

    os.makedirs(app.instance_path, exist_ok=True)

    app.config["SQLALCHEMY_DATABASE_URI"] = (
        "sqlite:///"
        + os.path.join(app.instance_path, "fiscal.db")
    )

    app.config["SQLALCHEMY_TRACK_MODIFICATIONS"] = False
    app.config["MAX_CONTENT_LENGTH"] = 5 * 1024 * 1024  # 5 Mo max par import

    if config:
        app.config.update(config)

    db.init_app(app)
    migrate.init_app(app, db)

    # Charger les modèles
    from app import models

    # Importer les Blueprints
    from app.accueil.routes import accueil_bp
    from app.suivi_impayes.routes import suivi_impayes_bp
    from app.recherche.routes import recherche_bp
    from app.synthese.routes import synthese_bp
    from app.rapports.routes import rapports_bp
    from app.importation.routes import importation_bp
    # Enregistrer les Blueprints
    app.register_blueprint(accueil_bp)
    app.register_blueprint(suivi_impayes_bp)
    app.register_blueprint(recherche_bp)
    app.register_blueprint(synthese_bp)
    app.register_blueprint(rapports_bp)
    app.register_blueprint(importation_bp)
    return app

from decimal import Decimal

from app.models import DossierFiscal, Paiement


def get_dashboard_stats():
    """Indicateurs du tableau de bord, calculés depuis la base."""
    dossiers = DossierFiscal.query.all()

    stats = {
        "total_dossiers": len(dossiers),
        "dossiers_impayes": 0,
        "dossiers_partiels": 0,
        "dossiers_payes": 0,
        "en_retard": 0,
        "montant_restant": Decimal("0.00"),
    }

    for dossier in dossiers:
        stats["montant_restant"] += dossier.montant_restant

        if dossier.statut == "PAYE":
            stats["dossiers_payes"] += 1
        elif dossier.statut == "PARTIELLEMENT_PAYE":
            stats["dossiers_partiels"] += 1
        else:
            stats["dossiers_impayes"] += 1

        if dossier.jours_retard > 0:
            stats["en_retard"] += 1

    return stats


def get_recent_activities(limit=10):
    """Derniers paiements enregistrés (en attendant le journal d'audit)."""
    limit = max(limit, 1)

    paiements = (
        Paiement.query
        .order_by(Paiement.date_enregistrement.desc())
        .limit(limit)
        .all()
    )

    return [
        {
            "date": p.date_enregistrement.strftime("%d/%m/%Y %H:%M"),
            "operation": "Paiement enregistré",
            "reference": p.dossier.numero_dossier,
            "statut": "Enregistré",
        }
        for p in paiements
    ]

import os

from flask import (
    Blueprint,
    render_template,
    request,
    flash,
    redirect,
    url_for,
)

from app.importation.services import importer_dossiers


importation_bp = Blueprint(
    "importation",
    __name__,
    url_prefix="/importation",
)


@importation_bp.route("/", methods=["GET", "POST"])
def index():

    resultat = None

    if request.method == "POST":

        fichier = request.files.get("fichier")

        if not fichier or fichier.filename == "":
            flash("Veuillez sélectionner un fichier.", "error")
            return redirect(url_for("importation.index"))

        extension = os.path.splitext(fichier.filename)[1].lower()

        if extension not in [".csv", ".xlsx"]:
            flash("Format non accepté. Utilisez CSV ou Excel.", "error")
            return redirect(url_for("importation.index"))

        try:
            resultat = importer_dossiers(
                fichier,
                fichier.filename,
            )

            flash(
                f"Importation terminée : {resultat['nombre_importes']} dossiers importés.",
                "success",
            )

        except Exception as erreur:
            flash(f"Erreur d'importation : {str(erreur)}", "error")

    return render_template(
        "importation/index.html",
        resultat=resultat,
    )

import pandas as pd
import json

from io import BytesIO
from datetime import datetime, date
from decimal import Decimal, InvalidOperation

from app.extensions import db
from app.models import (
    Contribuable,
    TypeTaxe,
    DossierFiscal,
    ImportBatch,
)


def lire_fichier(fichier, nom_fichier):
    """Lire un fichier CSV ou Excel."""

    extension = nom_fichier.rsplit(".", 1)[-1].lower()
    contenu = fichier.read()

    if extension == "csv":
        try:
            df = pd.read_csv(
                BytesIO(contenu),
                sep=None,
                engine="python",
                dtype=str,
            )
        except UnicodeDecodeError:
            df = pd.read_csv(
                BytesIO(contenu),
                sep=None,
                engine="python",
                encoding="latin-1",
                dtype=str,
            )

    elif extension == "xlsx":
        df = pd.read_excel(BytesIO(contenu), dtype=str)

    else:
        raise ValueError("Format non supporté. Utilisez CSV ou XLSX.")

    df.columns = [
        str(colonne).strip().lower()
        for colonne in df.columns
    ]

    return df


def convertir_date(valeur):
    """Convertir une valeur en date."""

    if pd.isna(valeur):
        raise ValueError("Date limite manquante.")

    if isinstance(valeur, (datetime, date)):
        return valeur.date() if isinstance(valeur, datetime) else valeur

    try:
        return pd.to_datetime(
            valeur,
            dayfirst=True,
            errors="raise",
        ).date()
    except Exception:
        raise ValueError("Date limite invalide.")


def convertir_montant(valeur):
    """Convertir un montant en Decimal."""

    if pd.isna(valeur):
        raise ValueError("Montant exigible manquant.")

    try:
        montant = Decimal(str(valeur).replace(",", "."))
    except (InvalidOperation, ValueError):
        raise ValueError("Montant exigible invalide.")

    if not montant.is_finite() or montant <= 0:
        raise ValueError("Le montant exigible doit être positif.")

    return montant


def importer_dossiers(fichier, nom_fichier):
    """Importer les dossiers fiscaux depuis CSV ou Excel."""

    extension = nom_fichier.rsplit(".", 1)[-1].lower()

    batch = ImportBatch(
        nom_fichier=nom_fichier,
        type_fichier=extension.upper(),
        statut="EN_COURS",
    )

    db.session.add(batch)
    db.session.flush()

    erreurs = []
    nombre_importes = 0

    try:
        df = lire_fichier(fichier, nom_fichier)

        colonnes_requises = [
            "numero_dossier",
            "identifiant_fiscal",
            "type_taxe",
            "annee_fiscale",
            "date_limite",
            "montant_exigible",
        ]

        colonnes_manquantes = [
            colonne
            for colonne in colonnes_requises
            if colonne not in df.columns
        ]

        if colonnes_manquantes:
            raise ValueError(
                "Colonnes manquantes : "
                + ", ".join(colonnes_manquantes)
            )

        batch.nombre_lignes = len(df)

        for index, ligne in df.iterrows():
            numero_ligne = index + 2

            try:
                with db.session.begin_nested():

                    numero_dossier = str(
                        ligne["numero_dossier"]
                    ).strip()

                    identifiant_fiscal = str(
                        ligne["identifiant_fiscal"]
                    ).strip()

                    type_taxe_libelle = str(
                        ligne["type_taxe"]
                    ).strip()

                    if not numero_dossier or numero_dossier.lower() == "nan":
                        raise ValueError("Numéro de dossier manquant.")

                    if not identifiant_fiscal or identifiant_fiscal.lower() == "nan":
                        raise ValueError("Identifiant fiscal manquant.")

                    if not type_taxe_libelle or type_taxe_libelle.lower() == "nan":
                        raise ValueError("Type de taxe manquant.")

                    contribuable = Contribuable.query.filter_by(
                        identifiant_fiscal=identifiant_fiscal
                    ).first()

                    if not contribuable:
                        raise ValueError(
                            f"Contribuable introuvable : {identifiant_fiscal}"
                        )

                    type_taxe = TypeTaxe.query.filter_by(
                        libelle=type_taxe_libelle
                    ).first()

                    if not type_taxe:
                        raise ValueError(
                            f"Type de taxe introuvable : {type_taxe_libelle}"
                        )

                    existe = DossierFiscal.query.filter_by(
                        numero_dossier=numero_dossier
                    ).first()

                    if existe:
                        raise ValueError(
                            f"Dossier déjà existant : {numero_dossier}"
                        )

                    try:
                        annee = int(float(str(ligne["annee_fiscale"]).strip()))
                    except ValueError:
                        raise ValueError("Année fiscale invalide.")

                    if annee < 1900 or annee > 2100:
                        raise ValueError("Année fiscale invalide.")

                    date_limite = convertir_date(
                        ligne["date_limite"]
                    )

                    montant = convertir_montant(
                        ligne["montant_exigible"]
                    )

                    dossier = DossierFiscal(
                        numero_dossier=numero_dossier,
                        contribuable_id=contribuable.id,
                        type_taxe_id=type_taxe.id,
                        annee_fiscale=annee,
                        date_limite=date_limite,
                        montant_exigible=montant,
                    )

                    db.session.add(dossier)

                nombre_importes += 1

            except Exception as erreur:
                erreurs.append({
                    "ligne": numero_ligne,
                    "erreur": str(erreur),
                })

        batch.nombre_importes = nombre_importes
        batch.nombre_erreurs = len(erreurs)

        batch.details_erreurs = json.dumps(
            erreurs,
            ensure_ascii=False,
        )

        batch.statut = (
            "TERMINE"
            if not erreurs
            else "TERMINE_AVEC_ERREURS"
        )

        db.session.commit()

        return {
            "batch": batch,
            "nombre_importes": nombre_importes,
            "nombre_erreurs": len(erreurs),
            "erreurs": erreurs,
        }

    except Exception:
        db.session.rollback()
        raise

from datetime import datetime

from app.extensions import db


class Contribuable(db.Model):
    __tablename__ = "contribuables"

    id = db.Column(db.Integer, primary_key=True)

    nom = db.Column(db.String(100), nullable=False)
    prenom = db.Column(db.String(100), nullable=False, default="")

    cin = db.Column(db.String(20), unique=True, nullable=True)

    identifiant_fiscal = db.Column(
        db.String(50),
        unique=True,
        nullable=False
    )

    telephone = db.Column(db.String(20), nullable=True)

    date_creation = db.Column(
        db.DateTime,
        default=datetime.utcnow,
        nullable=False
    )

    # Un contribuable peut avoir plusieurs dossiers fiscaux.
    dossiers = db.relationship(
        "DossierFiscal",
        back_populates="contribuable",
        lazy=True
    )

    @property
    def nom_complet(self):
        return f"{self.prenom} {self.nom}".strip()

    def __repr__(self):
        return f"<Contribuable {self.identifiant_fiscal}>"

        "Paiement",
        back_populates="dossier",
        cascade="all, delete-orphan",
        order_by="Paiement.date_paiement"
    )

    relances = db.relationship(
        "Relance",
        back_populates="dossier",
        cascade="all, delete-orphan",
        order_by="Relance.date_tentative.desc()"
    )

    @property
    def montant_paye(self):
        return sum(
            (paiement.montant for paiement in self.paiements),
            Decimal("0.00")
        )

    @property
    def montant_restant(self):
        restant = Decimal(self.montant_exigible) - self.montant_paye
        return max(restant, Decimal("0.00"))

    @property
    def statut(self):
        if self.montant_restant == Decimal("0.00"):
            return "PAYE"

        if self.montant_paye > Decimal("0.00"):
            return "PARTIELLEMENT_PAYE"

        return "IMPAYE"

    @property
    def jours_retard(self):
        if self.montant_restant <= Decimal("0.00"):
            return 0

        return max((date.today() - self.date_limite).days, 0)

    def __repr__(self):
        return f"<DossierFiscal {self.numero_dossier}>"

def obtenir_dossiers():
    """Récupère les dossiers fiscaux avec leurs contribuables et taxes."""
    return (
        DossierFiscal.query
        .join(DossierFiscal.contribuable)
        .join(DossierFiscal.type_taxe)
        .order_by(DossierFiscal.date_limite.asc())
        .all()
    )


def preparer_donnees():
    """Prépare les lignes communes aux exports PDF et Excel."""
    dossiers = obtenir_dossiers()

    lignes = []

    for dossier in dossiers:
        lignes.append([
            dossier.numero_dossier,
            dossier.contribuable.nom_complet,
            dossier.contribuable.identifiant_fiscal or "",
            dossier.type_taxe.libelle,
            dossier.annee_fiscale,
            float(dossier.montant_exigible),
            float(dossier.montant_paye),
            float(dossier.montant_restant),
            dossier.date_limite.strftime("%d/%m/%Y"),
            dossier.jours_retard,
            {
                "PAYE": "Payé",
                "PARTIELLEMENT_PAYE": "Partiellement payé",
                "IMPAYE": "Impayé",
            }.get(dossier.statut, dossier.statut),
        ])

    return lignes


def generer_excel():
    """Génère un fichier Excel contenant les dossiers fiscaux."""
    workbook = Workbook()
    feuille = workbook.active
    feuille.title = "Dossiers fiscaux"

    entetes = [
        "Référence",
        "Contribuable",
        "Identifiant fiscal",
        "Type de taxe",
        "Année fiscale",
        "Montant exigible (MAD)",
        "Montant payé (MAD)",
        "Montant restant (MAD)",
        "Date limite",
        "Jours de retard",
        "Statut",
    ]

    feuille.append(["RAPPORT DES DOSSIERS FISCAUX"])
    feuille.merge_cells(start_row=1, start_column=1,
                        end_row=1, end_column=len(entetes))
    feuille["A1"].font = Font(bold=True, size=16, color="FFFFFF")
    feuille["A1"].fill = PatternFill(
        "solid", fgColor="17365D"
    )
    feuille["A1"].alignment = Alignment(horizontal="center")

    feuille.append(entetes)

    for cellule in feuille[2]:
        cellule.font = Font(bold=True, color="FFFFFF")
        cellule.fill = PatternFill("solid", fgColor="24558A")
        cellule.alignment = Alignment(
            horizontal="center",
            vertical="center",
            wrap_text=True,
        )

    for ligne in preparer_donnees():
        feuille.append(ligne)

    for ligne in feuille.iter_rows(min_row=3, min_col=6, max_col=8):
        for cellule in ligne:
            cellule.number_format = '#,##0.00 "MAD"'

    largeurs = [21, 25, 22, 20, 15, 23, 20, 23, 16, 17, 24]

    for index, largeur in enumerate(largeurs, start=1):
        feuille.column_dimensions[
            get_column_letter(index)
        ].width = largeur

    feuille.freeze_panes = "A3"
    feuille.auto_filter.ref = (
        f"A2:K{feuille.max_row}"
    )

    sortie = BytesIO()
    workbook.save(sortie)
    sortie.seek(0)

    return sortie

            Contribuable.identifiant_fiscal.ilike(

            )

            dossiers_query = dossiers_query.filter(
                DossierFiscal.type_taxe_id == int(type_taxe_id)
            )
        except ValueError:
            type_taxe_id = ""

    if annee:
        try:
            dossiers_query = dossiers_query.filter(
                DossierFiscal.annee_fiscale == int(annee)
            )
        except ValueError:
            annee = ""

    dossiers = dossiers_query.order_by(
        DossierFiscal.date_limite.asc()
    ).all()

    # Le statut et le retard sont calculés par le modèle.
    if statut in ("IMPAYE", "PARTIELLEMENT_PAYE", "PAYE"):
        dossiers = [
            d for d in dossiers if d.statut == statut
        ]

    if retard == "en_retard":
        dossiers = [
            d for d in dossiers
            if d.montant_restant > 0 and d.jours_retard > 0
        ]
    elif retard == "sans_retard":
        dossiers = [
            d for d in dossiers
            if d.montant_restant > 0 and d.jours_retard == 0
        ]

    types_taxes = TypeTaxe.query.order_by(
        TypeTaxe.libelle.asc()
    ).all()

    annees = sorted(
        {d.annee_fiscale for d in DossierFiscal.query.all()},
        reverse=True,
    )

    return render_template(

        query=query,
        identifiant=identifiant,
        type_taxe_id=type_taxe_id,
        annee=annee,

"""Logique métier des paiements (règles du cahier des charges, §4.5 et §5.4)."""

from datetime import date
from decimal import Decimal, InvalidOperation

from app.extensions import db
from app.models import Paiement


class PaiementInvalide(ValueError):
    """Erreur de validation d'un paiement (message lisible par l'utilisateur)."""


def convertir_montant_paiement(valeur):
    try:
        montant = Decimal(str(valeur).strip().replace(",", "."))
    except (InvalidOperation, ValueError):
        raise PaiementInvalide("Montant invalide.")

    if not montant.is_finite() or montant <= 0:
        raise PaiementInvalide("Le montant du paiement doit être strictement positif.")

    return montant.quantize(Decimal("0.01"))


def enregistrer_paiement(dossier, montant, date_paiement=None, reference=None, observations=None):
    """Valide puis enregistre un paiement dans une seule transaction.

    Refuse un montant <= 0 ou supérieur au montant restant.
    En cas d'erreur, rien n'est enregistré.
    """
    if dossier is None:
        raise PaiementInvalide("Le paiement doit être associé à un dossier existant.")

    montant = convertir_montant_paiement(montant)

    if montant > dossier.montant_restant:
        raise PaiementInvalide(
            f"Le paiement ({montant}) dépasse le montant restant ({dossier.montant_restant})."
        )

    reference = (reference or "").strip() or None
    if reference and Paiement.query.filter_by(reference=reference).first():
        raise PaiementInvalide(f"La référence de paiement existe déjà : {reference}")

    paiement = Paiement(
        dossier_id=dossier.id,
        montant=montant,
        date_paiement=date_paiement or date.today(),
        reference=reference,
        observations=observations,
    )

    try:
        db.session.add(paiement)
        db.session.commit()
    except Exception:
        db.session.rollback()
        raise

    return paiement

from decimal import Decimal

from app.extensions import db
from app.models import DossierFiscal


def obtenir_statistiques():
    """Calcule les indicateurs financiers à partir des dossiers fiscaux."""

    dossiers = DossierFiscal.query.all()

    nombre_dossiers = len(dossiers)

    nombre_payes = sum(
        1 for dossier in dossiers
        if dossier.statut == "PAYE"
    )

    nombre_impayes = sum(
        1 for dossier in dossiers
        if dossier.statut == "IMPAYE"
    )

    nombre_partiels = sum(
        1 for dossier in dossiers
        if dossier.statut == "PARTIELLEMENT_PAYE"
    )

    montant_exigible = sum(
        (Decimal(str(d.montant_exigible)) for d in dossiers),
        Decimal("0.00")
    )

    montant_paye = sum(
        (Decimal(str(d.montant_paye)) for d in dossiers),
        Decimal("0.00")
    )

    montant_restant = sum(
        (Decimal(str(d.montant_restant)) for d in dossiers),
        Decimal("0.00")
    )

    nombre_en_retard = sum(
        1 for dossier in dossiers
        if dossier.montant_restant > 0
        and dossier.jours_retard > 0
    )

    taux_recouvrement = (
        float(montant_paye / montant_exigible * 100)
        if montant_exigible > 0
        else 0
    )

    return {
        "nombre_dossiers": nombre_dossiers,
        "nombre_payes": nombre_payes,
        "nombre_impayes": nombre_impayes,
        "nombre_partiels": nombre_partiels,
        "nombre_en_retard": nombre_en_retard,
        "montant_exigible": montant_exigible,
        "montant_paye": montant_paye,
        "montant_restant": montant_restant,
        "taux_recouvrement": taux_recouvrement,
    }

from datetime import date, timedelta
from decimal import Decimal

from app import create_app
from app.extensions import db
from app.models import Contribuable, DossierFiscal, Paiement, TypeTaxe

TYPES_TAXES = [
    ("TVA", "Taxe sur la valeur ajoutée"),
    ("IR", "Impôt sur le revenu"),
    ("IS", "Impôt sur les sociétés"),
    ("TP", "Taxe professionnelle"),
]

CONTRIBUABLES = [
    ("IF1000001", "Alaoui", "Karim"),
    ("IF1000002", "Bennani", "Salma"),
    ("IF1000003", "Chraibi", "Youssef"),
    ("IF1000004", "Drissi", "Nadia"),
    ("IF1000005", "El Idrissi", "Omar"),
]


def seed():
    if Contribuable.query.first():
        print("La base contient déjà des données : seed annulé.")
        return

    types = {}
    for code, libelle in TYPES_TAXES:
        t = TypeTaxe(code=code, libelle=libelle)
        db.session.add(t)
        types[code] = t

    contribuables = []
    for identifiant, nom, prenom in CONTRIBUABLES:
        c = Contribuable(identifiant_fiscal=identifiant, nom=nom, prenom=prenom)
        db.session.add(c)
        contribuables.append(c)

    db.session.flush()

    today = date.today()
    # (contribuable, taxe, année, échéance (jours / aujourd'hui), exigible, payé)
    cas = [
        (0, "TVA", 2025, -90, "12000.00", "0.00"),        # impayé, en retard
        (1, "IR", 2025, -45, "8000.00", "3000.00"),       # partiel, en retard
        (2, "IS", 2025, -10, "25000.00", "25000.00"),     # payé
        (3, "TP", 2026, 30, "4500.00", "0.00"),           # impayé, pas encore échu
        (4, "TVA", 2026, 0, "6000.00", "0.00"),           # échéance aujourd'hui : pas en retard
        (0, "IR", 2026, -200, "15000.00", "5000.00"),     # partiel, très en retard
    ]

    for i, (idx, code, annee, delta, exigible, paye) in enumerate(cas, start=1):
        dossier = DossierFiscal(
            numero_dossier=f"DEMO-{annee}-{i:04d}",
            contribuable_id=contribuables[idx].id,
            type_taxe_id=types[code].id,
            annee_fiscale=annee,
            date_limite=today + timedelta(days=delta),
            montant_exigible=Decimal(exigible),
        )
        db.session.add(dossier)
        db.session.flush()

        if Decimal(paye) > 0:
            db.session.add(Paiement(
                dossier_id=dossier.id,
                montant=Decimal(paye),
                date_paiement=today - timedelta(days=5),
                reference=f"PAY-DEMO-{i:04d}",
            ))

    db.session.commit()
    print("Données de démonstration créées.")


if __name__ == "__main__":
    from sqlalchemy.exc import OperationalError

    app = create_app()
    with app.app_context():
        try:
            seed()
        except OperationalError:
            db.session.rollback()
            print("Tables introuvables. Lancez d'abord : flask --app run db upgrade")

from datetime import date, timedelta
from decimal import Decimal

import pytest

from app.extensions import db
from app.models import Contribuable, DossierFiscal, TypeTaxe
from app.services.calculs_fiscaux import PaiementInvalide, enregistrer_paiement


def creer_dossier(delta_jours=-10, montant="1000.00"):
    c = Contribuable(identifiant_fiscal="IF-T1", nom="Test")
    t = TypeTaxe(code="T", libelle="Taxe test")
    db.session.add_all([c, t])
    db.session.flush()
    d = DossierFiscal(
        numero_dossier="D-1",
        contribuable_id=c.id,
        type_taxe_id=t.id,
        annee_fiscale=2025,
        date_limite=date.today() + timedelta(days=delta_jours),
        montant_exigible=Decimal(montant),
    )
    db.session.add(d)
    db.session.commit()
    return d


def test_impaye_en_retard(app):
    d = creer_dossier(-10)
    assert d.statut == "IMPAYE"
    assert d.montant_restant == Decimal("1000.00")
    assert d.jours_retard == 10


def test_echeance_aujourdhui_pas_en_retard(app):
    assert creer_dossier(0).jours_retard == 0


def test_paiement_partiel_puis_total(app):
    d = creer_dossier()
    enregistrer_paiement(d, "400")
    assert d.statut == "PARTIELLEMENT_PAYE"
    assert d.montant_restant == Decimal("600.00")
    enregistrer_paiement(d, "600,00")
    assert d.statut == "PAYE"
    assert d.jours_retard == 0


@pytest.mark.parametrize("valeur", ["0", "-5", "abc", ""])
def test_paiement_invalide_refuse(app, valeur):
    d = creer_dossier()
    with pytest.raises(PaiementInvalide):
        enregistrer_paiement(d, valeur)
    assert d.montant_paye == Decimal("0.00")


def test_paiement_superieur_au_restant_refuse(app):
    d = creer_dossier()
    with pytest.raises(PaiementInvalide):
        enregistrer_paiement(d, "1000.01")
    assert len(d.paiements) == 0


def test_reference_paiement_unique(app):
    d = creer_dossier()
    enregistrer_paiement(d, "100", reference="R1")
    with pytest.raises(PaiementInvalide):
        enregistrer_paiement(d, "100", reference="R1")

from io import BytesIO

from app.extensions import db
from app.models import Contribuable, DossierFiscal, TypeTaxe


def test_import_csv_zero_initial_et_erreurs(app):
    db.session.add_all([
        Contribuable(identifiant_fiscal="00123", nom="A"),
        TypeTaxe(code="TVA", libelle="TVA"),
    ])
    db.session.commit()

    csv = (
        "numero_dossier,identifiant_fiscal,type_taxe,annee_fiscale,date_limite,montant_exigible\n"
        "D1,00123,TVA,2025,31/12/2025,\"1500,50\"\n"
        "D2,99999,TVA,2025,31/12/2025,100\n"
        "D1,00123,TVA,2025,31/12/2025,100\n"
    )
    r = app.test_client().post(
        "/importation/",
        data={"fichier": (BytesIO(csv.encode()), "t.csv")},
        content_type="multipart/form-data",
    )
    assert r.status_code == 200
    assert DossierFiscal.query.count() == 1          # 1 valide, 2 rejetées
    assert str(DossierFiscal.query.first().montant_exigible) == "1500.50"
